
#include <Wire.h>
#include <Adafruit_PWMServoDriver.h>

Adafruit_PWMServoDriver pwm(0x40);

#define SERVOMIN 130
#define SERVOMAX 480

#define BASE     4
#define SHOULDER 5
#define ELBOW    6
#define GRIPPER  7

int baseHome = 95;
int shoulderHome = 120;
int elbowHome = 45;
int gripperOpen = 90;
int gripperClosed = 60;

int currentAngle[4] = {95, 120, 45, 90};

int angleToPulse(int angle) {
  return map(angle, 0, 180, SERVOMIN, SERVOMAX);
}

void setServo(int channel, int angle) {
  angle = constrain(angle, 0, 180);
  pwm.setPWM(channel, 0, angleToPulse(angle));
}

void moveArm(int channel, int targetAngle, int index) {
  targetAngle = constrain(targetAngle, 0, 180);

  int startAngle = currentAngle[index];

  if (targetAngle > startAngle) {
    for (int a = startAngle; a <= targetAngle; a++) {
      setServo(channel, a);
      delay(20);
    }
  } else {
    for (int a = startAngle; a >= targetAngle; a--) {
      setServo(channel, a);
      delay(20);
    }
  }

  currentAngle[index] = targetAngle;
}

void goHome() {
  moveArm(BASE, baseHome, 0);
  moveArm(SHOULDER, shoulderHome, 1);
  moveArm(ELBOW, elbowHome, 2);
  moveArm(GRIPPER, gripperOpen, 3);
}

void pickAndDrop(bool dropRight) {
  // Open gripper
  moveArm(GRIPPER, gripperOpen, 3);
  delay(500);

  // Lower arm
  moveArm(SHOULDER, 120, 1);
  moveArm(ELBOW, 60, 2);
  delay(500);

  // Close gripper
  moveArm(GRIPPER, gripperClosed, 3);
  delay(700);

  // Lift arm
  moveArm(SHOULDER, 100, 1);
  moveArm(ELBOW, 80, 2);
  delay(500);

  // Move towards drop side
  if (dropRight) {
    moveArm(BASE, 150, 0);
  } else {
    moveArm(BASE, 30, 0);
  }

  delay(500);

  // Release object
  moveArm(SHOULDER, 120, 1);
  moveArm(ELBOW, 60, 2);
  moveArm(GRIPPER, gripperOpen, 3);

  delay(500);

  // Return home
  goHome();
}

void setup() {
  Serial.begin(115200);

  Wire.begin(21, 22);
  pwm.begin();
  pwm.setPWMFreq(50);
  delay(500);

  goHome();

  Serial.println("Robotic Arm Ready");
  Serial.println("Commands: PICK_RED, PICK_BLUE, HOME");
}

void loop() {
  if (Serial.available()) {
    String command = Serial.readStringUntil('\n');
    command.trim();

    if (command == "PICK_RED") {
      Serial.println("Picking red object");
      pickAndDrop(true);
      Serial.println("Red pick sequence complete");
    }
    else if (command == "PICK_BLUE") {
      Serial.println("Picking blue object");
      pickAndDrop(false);
      Serial.println("Blue pick sequence complete");
    }
    else if (command == "HOME") {
      goHome();
      Serial.println("Arm at home");
    }
  }
}
